Deployment & Service connection
Deployment:
  metadata:
    labels:
      app: ski
  spec:
    template:
      metadata:
        labels:
          app: ski
    spec:
      containers:
        - name:
          ports:
            - containerPort: 8080
              port: 80

Service:
  metadata:
  spec:
    selector:
      app: ski
    ports:
      - port: 80
        targetPort: 8080
